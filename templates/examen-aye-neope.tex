\newcommand{\instruccionessc}[2]{
    \textsf{
     #1 minutos\\
     \vspace{-3mm}\hrule
     \textbf{Instrucciones para los alumnos}\\
     \begin{itemize}
         \item No abra esta prueba hasta que se lo autoricen.
         \item En esta prueba no se permite el uso de ninguna calculadora.
         \item Se debe responder a cada pregunta en el cuadro reservado para ello.
         \item Salvo que se indique lo contrario en la pregunta, todas las respuestas deben estar simplificadas.
         \item Se necesita una copia sin anotaciones del cuadernillo de fórmulas de \textbf{Matemáticas: Análisis y Enfoques} para esta prueba.
         \item Si se ve obligado a rectificar una respuesta, o no tiene suficiente con el espacio asignado, puede contestar en el cuadernillo adjunto, indicando el número de ejercicio y apartado. Deberá tachar obligatoriamente la respuesta incorrecta.
         \item La puntuación máxima para esta prueba de examen es \textbf{[#2 puntos]}.
     \end{itemize}}
    \newpage
}

\newcommand{\instruccionescc}[2]{

     \textbf{Instrucciones para los alumnos}\\
     \begin{itemize}
         \item No abra esta prueba hasta que se lo autoricen.
         \item En esta prueba es necesario utilizar una calculadora de pantalla gráfica.
         \item Se debe responder a cada pregunta en el cuadro reservado para ello.
         \item Salvo que se indique lo contrario en la pregunta, todas las respuestas numéricas deberán ser exactas o aproximadas con tres cifras significativas.
         \item Se necesita una copia sin anotaciones del cuadernillo de fórmulas de \textbf{Matemáticas: Análisis y Enfoques} para esta prueba.
         \item Si se ve obligado a rectificar una respuesta, o no tiene suficiente con el espacio asignado, puede contestar en el cuadernillo adjunto, indicando el número de ejercicio y apartado. Deberá tachar obligatoriamente la respuesta incorrecta.
         \item La puntuación máxima para esta prueba de examen es \textbf{[#2 puntos]}.
     \end{itemize}}
    \newpage
}
